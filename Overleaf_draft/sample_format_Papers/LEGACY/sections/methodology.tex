\section{Unified Training Objective}
\label{sec:unified_loss}

CITA combines three loss components for instruction-conditioned alignment:

\begin{equation}
\mathcal{L}_{\text{unified}} = \mathcal{L}_{\text{SFT}} + \lambda_1 \mathcal{L}_{\text{DPO}} + \lambda_2 \mathcal{L}_{\text{KL}}
\end{equation}

\noindent\textbf{SFT Loss} (response quality):
\begin{equation}
\mathcal{L}_{\text{SFT}} = -\frac{1}{N} \sum_{(x, y^+)} \sum_{t=1}^{T} \log P_\pi(y_t^+ | x, y_{<t}^+)
\end{equation}

\noindent\textbf{DPO Loss} (preference alignment):
\begin{equation}
\mathcal{L}_{\text{DPO}} = -\frac{1}{N} \sum \log \sigma \left( \beta \cdot \Delta \right)
\end{equation}
where $\Delta = \log P_\pi(y^+|x) - \log P_\pi(y^-|x)$.

\noindent\textbf{KL Loss} (stability, \textit{mandatory} in CITA):
\begin{equation}
\mathcal{L}_{\text{KL}} = \frac{1}{N} \sum_{(x,y)} \text{KL}[P_\pi \| P_{\text{ref}}]
\end{equation}

\section{CITA: Contrastive Instruction-Tuned Alignment}

\subsection{Training Triplet}

Let $I$ = instruction, $X$ = input, $Y^+$ = accepted, $Y^-$ = rejected.

\subsection{CITA-KL Loss}

\begin{equation}
\mathcal{L}_{\text{CITA}} = -\log P^+ + \lambda \cdot \text{KL}[\pi_\theta \| \pi_0]
\end{equation}

where:
\begin{equation}
P^+ = \frac{\exp(\beta s^+)}{\exp(\beta s^+) + \exp(\beta s^-)}
\end{equation}

and $s^+ = \log \pi_\theta(Y^+ | I, X)$, $s^- = \log \pi_\theta(Y^- | I, X)$.

\textbf{Parameters:}
\begin{itemize}
    \item $\pi_\theta$: fine-tuned model
    \item $\pi_0$: base model
    \item $\beta$: temperature
    \item $\lambda$: KL weight (\textbf{mandatory})
\end{itemize}

\subsection{Gradient}

\begin{equation}
\nabla_\theta \mathcal{L} = -\beta [(1-P^+)\nabla s^+ - P^+\nabla s^-]
\end{equation}

\subsection{Benefits}

\begin{itemize}
    \item \textbf{Generalization}: Novel prompts
    \item \textbf{Robustness}: Resists jailbreaks
    \item \textbf{Calibration}: Avoids over-refusal
    \item \textbf{Switching}: Adapts to instructions
\end{itemize}
